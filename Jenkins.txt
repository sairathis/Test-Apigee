pipeline {

    agent any

    tools {
        maven 'Maven'
    }

    environment {

        APIGEE_ORG = 'my-org'

        DEV_ENV = 'dev'
        QA_ENV = 'qa'
        STAGE_ENV = 'stage'
        PROD_ENV = 'prod'

    }

    stages {

        stage('Checkout') {
            steps {
                git branch: 'develop',
                url: 'https://bitbucket.org/company/customer-api.git'
            }
        }

        stage('ApigeeLint') {
            steps {
                sh '''
                npm install -g apigeelint
                apigeelint -s apiproxy
                '''
            }
        }

        stage('SonarQube Analysis') {
            steps {
                withSonarQubeEnv('SonarQube') {
                    sh '''
                    mvn sonar:sonar
                    '''
                }
            }
        }

        stage('Quality Gate') {
            steps {
                timeout(time: 10, unit: 'MINUTES') {
                    waitForQualityGate abortPipeline: true
                }
            }
        }

        stage('Build') {
            steps {

                sh '''
                zip -r customer-api.zip apiproxy
                '''
            }
        }

        stage('Deploy DEV') {
            steps {
                sh '''
                mvn apigee-enterprise:deploy \
                -Pdev \
                -Dapigee.org=${APIGEE_ORG} \
                -Dapigee.env=${DEV_ENV}
                '''
            }
        }

        stage('Integration Test') {
            steps {
                sh '''
                newman run tests/dev_collection.json
                '''
            }
        }

        stage('Deploy QA') {
            steps {
                sh '''
                mvn apigee-enterprise:deploy \
                -Pqa \
                -Dapigee.org=${APIGEE_ORG} \
                -Dapigee.env=${QA_ENV}
                '''
            }
        }

        stage('Regression Test') {
            steps {
                sh '''
                newman run tests/regression_suite.json
                '''
            }
        }

        stage('Deploy STAGE') {
            steps {
                sh '''
                mvn apigee-enterprise:deploy \
                -Pstage \
                -Dapigee.org=${APIGEE_ORG} \
                -Dapigee.env=${STAGE_ENV}
                '''
            }
        }

        stage('Performance Test') {
            steps {
                sh '''
                k6 run performance/load-test.js
                '''
            }
        }

        stage('Approval') {
            steps {
                input(
                    message: 'Approve Production Deployment?',
                    ok: 'Deploy'
                )
            }
        }

        stage('Deploy PROD') {
            steps {
                sh '''
                mvn apigee-enterprise:deploy \
                -Pprod \
                -Dapigee.org=${APIGEE_ORG} \
                -Dapigee.env=${PROD_ENV}
                '''
            }
        }

        stage('Smoke Test') {
            steps {
                sh '''
                newman run tests/smoke_test.json
                '''
            }
        }

    }

    post {

        success {
            echo 'Deployment Successful'
        }

        failure {
            echo 'Deployment Failed'
        }
    }
}

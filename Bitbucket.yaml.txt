  caches:
    sonar: ~/.sonar/cache

pipelines:

  branches:

    develop:
      - step:
          name: Checkout & ApigeeLint
          image: node:18
          caches:
            - node
          script:
            - npm install -g apigeelint
            - apigeelint -s apiproxy -f table.js

      - step:
          name: SonarQube Analysis
          caches:
            - sonar
          script:
            - mvn sonar:sonar
              -Dsonar.projectKey=customer-api
              -Dsonar.host.url=$SONAR_HOST
              -Dsonar.login=$SONAR_TOKEN

      - step:
          name: Build Proxy Bundle
          script:
            - zip -r customer-api.zip apiproxy/

            - customer-api.zip

          name: Deploy DEV
          script:
            - mvn clean install

                -Dapigee.env=dev

          name: Integration Test
          image: postman/newman
